# indices-monitor_V3

# CREER UN ENV VIRTUEL PYTHON 

    python -m venv mon

# ACTIVER L'ENVIRONEMENT VIRTUEL PYTHON WINDOWS

    mon\Scripts\Activate.bat

    source mon/bin/activate

    pip install -r requirements.txt
